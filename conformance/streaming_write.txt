# CX Conformance — streaming-write API
# Version: 0.3 — 2026-05-10 (JSON + YAML AST-shape emit, Phase 7.74j)
# Version: 0.2 — 2026-05-10 (XML emit + Tier-1 binding wrappers, Phase 7.74h)
# Version: 0.1 — 2026-05-10 (V core impl landed in Phase 7.74g)
# Covers: spec/streaming.md §6, ADR 0011, spec/abi.md §2.15.
#
# v0.6.0 scope (per spec/streaming.md §6.6 v0.3): CX + XML + JSON +
# YAML output formats implemented end-to-end. JSON and YAML use the
# **AST shape** locked in §6.6.1 (every event maps to its canonical
# AST node — same shape as `cx_to_ast_json`). TOML and MD return W009
# for the entire format (structural mismatch — §6.6.2); adopters use
# the non-streaming `cx_to_toml` / `cx_to_md` surfaces for those
# formats.
#
# Negative-validation fixtures (W001-W013) are exhaustive at v0.6.0 —
# adopters can rely on the W-code namespace as a stable contract.
#
# Fixture shape:
#   --- events
#       a sequence of newline-separated event statements. Each
#       statement is a single line with one of these forms:
#         StartDoc
#         EndDoc
#         StartElement <name> [anchor=<a>] [data_type=<t>] [merge=<m>]
#         EndElement   <name>
#         Text         "<value>"
#         Scalar       <type>:<value>
#         Comment      "<value>"
#         PI           <target> [data="<value>"]
#         EntityRef    <name>
#         RawText      "<value>"
#         Alias        <name>
#         StartTable   <col_spec_hex>
#         RowGroup     <payload_hex>
#         EndTable
#   --- format
#       cx | xml | json | yaml | toml | md
#   --- expect_ok
#       expected output bytes (CX text or hex form per format).
#   --- expect_err
#       expected W-code prefix (W001-W013); when present, expect_ok is absent.

# vary on trailing-newline / empty-items handling and is not pinned

        value: "hello"

# ── Wiring: in-memory writer round-trip via cx_events_writer_* ────────────────
#
# Runners that consume this file should:
#   1. Open a writer via cx_events_writer_open(format).
#   2. Iterate the `events` block, calling the matching emit symbol per
#      line (start_doc / end_doc / start_element_with_len / etc.).
#   3. On `expect_err`: assert the next emit returns a non-NULL diagnostic
#      whose return-value text starts with the expected W-code.
#   4. On `expect_ok`: assert close_get_bytes() returns exactly the
#      declared bytes (or a format-equivalent canonical form for non-cx
#      formats — equivalence rules to be specified when those formats
#      ship).
