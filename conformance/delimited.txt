# CX Conformance — delimited (CSV / TSV / PSV) tests
# Version: 1.0 — 2026-05-08
# Covers: spec/decisions/0001-delimited-conversion.md (ADR 0001)
#         and spec/conversions.md §8.
#
# Sections:
#   in_cx   + out_csv  : parse CX, emit CSV (RFC 4180 default)
#   in_cx   + out_tsv  : parse CX, emit TSV
#   in_cx   + out_psv  : parse CX, emit PSV
#   in_csv  + out_cx   : parse CSV, emit CX (auto-typing)
#   in_tsv  + out_cx   : parse TSV, emit CX
#   in_psv  + out_cx   : parse PSV, emit CX
#
# All emit fixtures use `\n` row terminators in expected text; runner
# normalizes the emitter's `\r\n` (RFC 4180 default) to `\n` before
# comparing.

=== test: dlm-001-table-typed
level: core
tags: delimited emit table
# Direct :table block — declared columns drive output 1:1.
--- in_cx
[users :table[name:string age:int active:bool]
  alice 30 true
  bob 25 false
]
--- out_csv
name,age,active
alice,30,true
bob,25,false

=== test: dlm-002-repeated-row
level: core
tags: delimited emit repeated-row
# Repeated-row mode (D2): three sibling <user> elements; columns are
# union of their attrs in first-occurrence order. +admin → "true";
# absence → empty cell.
--- in_cx
[users
  [user id=1 name=alice +admin]
  [user id=2 name=bob]
  [user id=3 name=carol +admin]
]
--- out_csv
id,name,admin
1,alice,true
2,bob,
3,carol,true

=== test: dlm-003-dotted-path
level: core
tags: delimited emit dotted-path
# Dotted-path mode (D2): single hierarchy → one row.
--- in_cx
[config
  [server host=localhost port=8080 +tls]
  [logging level=info format=json]
]
--- out_csv
server.host,server.port,server.tls,logging.level,logging.format
localhost,8080,true,info,json

=== test: dlm-004-quote-on-comma
level: core
tags: delimited emit quoting
# RFC 4180 D3: field with embedded comma → quoted.
--- in_cx
[t :table[name:string note:string]
  alice 'hello, world'
  bob ok
]
--- out_csv
name,note
alice,"hello, world"
bob,ok

=== test: dlm-005-quote-doubling
level: core
tags: delimited emit quoting
# Embedded `"` is doubled per RFC 4180.
--- in_cx
[t :table[name:string note:string]
  alice 'she said "hi"'
]
--- out_csv
name,note
alice,"she said ""hi"""

=== test: dlm-007-tsv
level: core
tags: delimited emit tsv
# TSV: comma in field doesn't trigger quoting.
--- in_cx
[t :table[name:string note:string]
  alice 'hello, world'
  bob ok
]
--- out_tsv
name	note
alice	hello, world
bob	ok

=== test: dlm-008-psv
level: core
tags: delimited emit psv
--- in_cx
[t :table[a b]
  x y
  p q
]
--- out_psv
a|b
x|y
p|q

=== test: dlm-009-parse-basic
level: core
tags: delimited parse autotype
# Auto-typing D5: age → :int, active → :bool. name stays untyped.
--- in_csv
name,age,active
alice,30,true
bob,25,false
--- out_cx
[table :table[name age:int active:bool]
  alice 30 true
  bob 25 false
]

=== test: dlm-010-parse-quoted-stays-string
level: core
tags: delimited parse quoting
# Quoted "30" must NOT auto-type to int — quoting is the explicit
# string signal per D5.
--- in_csv
name,age
alice,"30"
bob,"25"
--- out_cx
[table :table[name age]
  alice 30
  bob 25
]

=== test: dlm-011-parse-singlequote
level: core
tags: delimited parse quoting
# Single-quote acceptance D4.
--- in_csv
name,note
alice,'hello, world'
--- out_cx
[table :table[name note]
  alice 'hello, world'
]

=== test: dlm-012-parse-empty-cell
level: core
tags: delimited parse null
# Empty unquoted cell → null. Column still narrows to :int because
# null doesn't constrain.
--- in_csv
name,age
alice,30
bob,
--- out_cx
[table :table[name age:int]
  alice 30
  bob null
]

=== test: dlm-013-parse-crlf
level: core
tags: delimited parse line-ending
# CRLF row terminators accepted on input.
--- in_csv

1,2
3,4
--- out_cx
[table :table[a:int b:int]
  1 2
  3 4
]
